25:     /// Begin even when reusing a completed frame: an A → B → A switch must
26:     /// invalidate B without requiring another render of the cached A image.
27:     func begin() -> UUID {
28:         let ticket = UUID()
29:         current = ticket
30:         return ticket
31:     }
32: 
33:     func invalidate() { current = nil }

35:     func load(ticket: UUID,
36:               render: () async throws -> StudyPreviewOutput,
37:               convert: (StudyImage) -> DisplayImage?,
38:               publish: (Completion) -> Void) async {
39:         guard current == ticket, !Task.isCancelled else { return }
40:         do {
41:             let output = try await render()
42:             guard current == ticket, !Task.isCancelled else { return }
43:             let image = convert(output.image)
44:             guard current == ticket, !Task.isCancelled else { return }
45:             publish(.init(image: image, unsupportedPixels: image == nil ? 0 : output.image.unsupportedPixels,
46:                           backendLabel: image == nil ? nil : output.backendLabel,
47:                           message: image == nil ? "The rendered image could not be displayed." : nil))
48:         } catch {
49:             guard current == ticket, !Task.isCancelled, !(error is CancellationError),
50:                   (error as? StudyError) != .cancelled else { return }
51:             let message: String
52:             switch error as? StudyError {
53:             case .invalidCamera: message = "Move the camera above the panel and away from the subject."
54:             case .invalidEmitter: message = "Check the aiming direction of each active light."
55:             default: message = "Check the camera, light positions, and preview settings."
56:             }
57:             publish(.init(image: nil, unsupportedPixels: 0, backendLabel: nil, message: message))
58:         }
59:     }

28:             return r.isFinite && (0...1).contains(r) && rough.isFinite && (0.05...1).contains(rough) && f0.isFinite && (0...0.08).contains(f0)
29:         }
30:     }
31: 
32:     /// Returns π·BRDF, making a pure diffuse material equal to its reflectance.
33:     /// The caller supplies incident irradiance (lux), including N·L.
34:     fileprivate func response(normal n: SIMD3<Double>, light l: SIMD3<Double>, view v: SIMD3<Double>, nl: Double) -> Double {
35:         switch self {
36:         case .diffuse(let r): return r
37:         case .dielectric(let r, let rough, let f0):
38:             let nv = max(0, dot3(n, v))
39:             guard nv > 0, let h = normalized3(l + v) else { return 0 }
40:             let nh = min(1, max(0, dot3(n, h)))
41:             let vh = min(1, max(0, dot3(v, h)))
42:             let alpha = rough * rough, a2 = alpha * alpha
43:             let d = nh * nh * (a2 - 1) + 1
44:             let distributionTimesPi = a2 / (d * d)
45:             // Height-correlated Smith visibility: stable at grazing angles.
46:             let denominator = 2 * (nl * sqrt(nv * nv * (1 - a2) + a2) + nv * sqrt(nl * nl * (1 - a2) + a2))
47:             let fresnel = f0 + (1 - f0) * pow(1 - vh, 5)
48:             let specular = denominator > 0 ? distributionTimesPi * fresnel / denominator : 0
49:             return r * (1 - fresnel) + specular
50:         }
51:     }
52: }

54: /// Diagnostic geometry: neutral sphere resting on a finite horizontal reference panel.
55: /// This plane is at subject height minus radius; it is not the actual room floor.
56: public struct StudyGeometry: Equatable, Sendable {
57:     public let center: Position
58:     public let radius: Double
59:     public let panelHalfExtent: Double
60:     public var panelY: Double { center.y - radius }
61: 
62:     public init?(center: Position, radius: Double = 0.22, panelHalfExtent: Double = 3) {
63:         guard center.isValid, radius.isFinite, (0.05...1).contains(radius),
64:               panelHalfExtent.isFinite, (1...10).contains(panelHalfExtent), panelHalfExtent >= radius else { return nil }
65:         self.center = center; self.radius = radius; self.panelHalfExtent = panelHalfExtent
66:     }
67: 
68:     fileprivate func inPanel(_ p: SIMD3<Double>) -> Bool {
69:         abs(p.x - center.x) <= panelHalfExtent && abs(p.z - center.z) <= panelHalfExtent
70:     }

72:     /// Positive ray/sphere intersection; a receiver's shadow ray is offset separately.
73:     fileprivate func sphereHit(origin: SIMD3<Double>, direction: SIMD3<Double>, minimum: Double) -> Double? {
74:         let offset = origin - vec(center)
75:         let b = dot3(offset, direction), c = dot3(offset, offset) - radius * radius
76:         let discriminant = b * b - c
77:         guard discriminant >= 0 else { return nil }
78:         let root = sqrt(discriminant)
79:         let near = -b - root, far = -b + root
80:         if near >= minimum { return near }
81:         if far >= minimum { return far }
82:         return nil
83:     }
84: 
85:     fileprivate func occludes(from origin: SIMD3<Double>, to direction: SIMD3<Double>, distance: Double) -> Bool {
86:         let epsilon = 1e-5
87:         if let hit = sphereHit(origin: origin, direction: direction, minimum: epsilon), hit < distance - epsilon { return true }
88:         if abs(direction.y) > 1e-12 {
89:             let t = (panelY - origin.y) / direction.y
90:             if t >= epsilon, t < distance - epsilon, inPanel(origin + direction * t) { return true }
91:         }
92:         return false
93:     }

104: /// Positive receiver incidence in normalized emitter coordinates [-1, 1]².
105: struct StudyAreaSamplingPlane {
106:     let center: Double
107:     let x: Double
108:     let y: Double
109:     var isFull: Bool { center >= abs(x) + abs(y) }
110:     var isEmpty: Bool { center <= -abs(x) - abs(y) }
111:     func at(_ p: SIMD2<Double>) -> Double { center + x * p.x + y * p.y }

114: /// A fixed number of samples over a moving convex domain. Clip the emitter once,
115: /// divide its x projection into equal columns, and use each column's exact area
116: /// and centroid x. Uniform vertical samples stay inside the illuminated interval.
117: /// Column areas and centroids vary continuously when a clip crosses a grid edge;
118: /// at the full-face limit the positions/weights approach the original midpoints.
119: /// Visibility can still have the usual discrete shadow-sampling transitions.
120: enum StudyAreaSliceSampling {
121:     static func forEach(grid: Int, normal: StudyAreaSamplingPlane, geometric: StudyAreaSamplingPlane? = nil,
122:                         _ body: (SIMD2<Double>, Double) throws -> Void) rethrows {
123:         guard !normal.isEmpty, !(geometric?.isEmpty ?? false) else { return }
124:         // A rectangle clipped by two hemisphere and two column planes has at
125:         // most eight vertices. Four separate buffers avoid input/output aliasing.
126:         try withUnsafeTemporaryAllocation(of: SIMD2<Double>.self, capacity: 32) { storage in
127:             let first = UnsafeMutableBufferPointer(rebasing: storage[0..<8])
128:             let second = UnsafeMutableBufferPointer(rebasing: storage[8..<16])
129:             let left = UnsafeMutableBufferPointer(rebasing: storage[16..<24])
130:             let column = UnsafeMutableBufferPointer(rebasing: storage[24..<32])
131:             first[0] = .init(-1, -1); first[1] = .init(1, -1)
132:             first[2] = .init(1, 1); first[3] = .init(-1, 1)
133:             var count = clip(first, count: 4, plane: normal, into: second)
134:             var polygon = second
135:             if let geometric {
136:                 count = clip(second, count: count, plane: geometric, into: first)
137:                 polygon = first
138:             }
139:             guard count >= 3 else { return }
140:             var minimum = polygon[0].x, maximum = minimum
141:             for i in 1..<count { minimum = min(minimum, polygon[i].x); maximum = max(maximum, polygon[i].x) }
142:             guard minimum < maximum else { return }
143:             for col in 0..<grid {
144:                 let a = minimum + (maximum - minimum) * Double(col) / Double(grid)
145:                 let b = minimum + (maximum - minimum) * Double(col + 1) / Double(grid)
146:                 let leftCount = clip(polygon, count: count, plane: .init(center: -a, x: 1, y: 0), into: left)
147:                 let columnCount = clip(left, count: leftCount, plane: .init(center: b, x: -1, y: 0), into: column)
148:                 guard columnCount >= 3 else { continue }
149:                 let anchor = column[0]
150:                 var twiceArea = 0.0, momentX = 0.0
151:                 for i in 1..<(columnCount - 1) {
152:                     let u = column[i] - anchor, v = column[i + 1] - anchor
153:                     let cross = abs(u.x * v.y - u.y * v.x)
154:                     twiceArea += cross
155:                     momentX += cross * (u.x + v.x) / 3
156:                 }
157:                 guard twiceArea > 0 else { continue }
158:                 let x = min(b, max(a, anchor.x + momentX / twiceArea))
159:                 var lower = -1.0, upper = 1.0
160:                 func constrain(_ plane: StudyAreaSamplingPlane) {
161:                     let atX = plane.center + plane.x * x
162:                     if plane.y > 0 { lower = max(lower, -atX / plane.y) }
163:                     else if plane.y < 0 { upper = min(upper, -atX / plane.y) }
164:                     else if atX <= 0 { upper = lower }
165:                 }
166:                 constrain(normal)
167:                 if let geometric { constrain(geometric) }
168:                 guard lower < upper else { continue }
169:                 // twiceArea / 8 is the column's fraction of the whole emitter.
170:                 // Each row takes 1/grid of it; base samples carry candela/grid².
171:                 let weight = twiceArea * Double(grid) / 8
172:                 for row in 0..<grid {
173:                     let y = lower + (upper - lower) * (Double(row) + 0.5) / Double(grid)
174:                     try body(.init(x, y), weight)
175:                 }
176:             }
177:         }
178:     }
179: 
180:     private static func clip(_ input: UnsafeMutableBufferPointer<SIMD2<Double>>, count: Int,
181:                              plane: StudyAreaSamplingPlane, into output: UnsafeMutableBufferPointer<SIMD2<Double>>) -> Int {
182:         guard count > 0 else { return 0 }
183:         var result = 0, previous = input[count - 1], previousDistance = plane.at(previous)
184:         for i in 0..<count {
185:             let current = input[i], distance = plane.at(current)
186:             // Intersect only strict sign crossings and retain exact tangencies.
187:             // Rounding can produce coincident vertices; their triangles have
188:             // zero area in the column moment calculation.
189:             if (previousDistance < 0 && distance > 0) || (previousDistance > 0 && distance < 0) {
190:                 let t = previousDistance / (previousDistance - distance)
191:                 output[result] = previous + (current - previous) * t
192:                 result += 1
193:             }
194:             if distance >= 0 { output[result] = current; result += 1 }
195:             previous = current; previousDistance = distance
196:         }
197:         return result
198:     }
199: }

201: /// Prepared light samples are shared across surface evaluations. Rectangular sources
202: /// retain midpoint samples on full faces and redistribute them continuously on
203: /// clipped faces, with weights proportional to the actual illuminated area.
204: public struct StudyLighting: Sendable {
205:     private struct Sample: Sendable {
206:         let position: SIMD3<Double>
207:         let emittingNormal: SIMD3<Double>?
208:         let candela: Double
209:         let xyz: SIMD3<Double>
210:         let areaIndex: Int?

212:     private struct AreaGuard: Sendable {
213:         let center: SIMD3<Double>
214:         let right: SIMD3<Double>
215:         let up: SIMD3<Double>
216:         let halfWidth: Double
217:         let halfHeight: Double
218:         let minimumDistance: Double
219: 
220:         func containsNearReceiver(_ p: SIMD3<Double>) -> Bool {
221:             let delta = p - center
222:             let x = min(halfWidth, max(-halfWidth, dot3(delta, right)))
223:             let y = min(halfHeight, max(-halfHeight, dot3(delta, up)))
224:             let separation = delta - right * x - up * y
225:             return dot3(separation, separation) < minimumDistance * minimumDistance
226:         }
227:     }
228:     private let samples: [Sample]
229:     private let areaGuards: [AreaGuard]
230:     private let grid: Int

232:     public init(scene: LightingScene, quality: StudyQuality = .interactive) throws {
233:         guard (try? scene.validated()) != nil else { throw StudyError.invalidScene }
234:         var result: [Sample] = []
235:         var guards: [AreaGuard] = []
236:         for light in scene.lights where light.candela > 0 {
237:             guard let color = PreviewColorimetry.blackbody(kelvin: light.kelvin) else { throw StudyError.invalidEmitter }
238:             let xyz = SIMD3<Double>(color.x, color.y, color.z)
239:             let optics = light.effectiveOptics, center = vec(light.position)
240:             if optics.kind == .point {
241:                 result.append(.init(position: center, emittingNormal: nil, candela: light.candela, xyz: xyz, areaIndex: nil))
242:             } else {
243:                 guard let frame = StudioPhotometry.frame(for: light, subject: scene.subject) else { throw StudyError.invalidEmitter }
244:                 let axis = vec(frame.normal)
245:                 if optics.kind == .directional {
246:                     result.append(.init(position: center, emittingNormal: axis, candela: light.candela, xyz: xyz, areaIndex: nil))
247:                 } else {
248:                     let grid = quality.grid, right = vec(frame.right), up = vec(frame.up)
249:                     let perCell = light.candela / Double(grid * grid)
250:                     let areaIndex = guards.count
251:                     guards.append(.init(center: center, right: right, up: up,
252:                                         halfWidth: optics.widthMeters / 2, halfHeight: optics.heightMeters / 2,
253:                                         minimumDistance: max(0.05, max(optics.widthMeters, optics.heightMeters) / 4)))
254:                     for row in 0..<grid {
255:                         let y = ((Double(row) + 0.5) / Double(grid) - 0.5) * optics.heightMeters
256:                         for col in 0..<grid {
257:                             let x = ((Double(col) + 0.5) / Double(grid) - 0.5) * optics.widthMeters
258:                             result.append(.init(position: center + right * x + up * y, emittingNormal: axis,
259:                                                 candela: perCell, xyz: xyz, areaIndex: areaIndex))
260:                         }
261:                     }
262:                 }
263:             }
264:         }
265:         samples = result; areaGuards = guards; grid = quality.grid
266:     }

270:     // Structure-of-vectors contract: three aligned float4 rows per sample and area.
271:     // Translate positions to the subject center to reduce Float32 roundoff.
272:     fileprivate func computeSamples(relativeTo center: SIMD3<Double>) -> [SIMD4<Float>] {
273:         samples.flatMap { s in
274:             [pack(s.position - center, s.candela), pack(s.emittingNormal ?? .zero, s.emittingNormal == nil ? 0 : 1),
275:              pack(s.xyz, s.areaIndex.map { Double($0 + 1) } ?? 0)]
276:         }
277:     }

279:     fileprivate func computeAreas(relativeTo center: SIMD3<Double>) -> [SIMD4<Float>] {
280:         areaGuards.flatMap { a in
281:             [pack(a.center - center, a.minimumDistance), pack(a.right, a.halfWidth), pack(a.up, a.halfHeight)]
282:         }

285:     /// Evaluates one arbitrary receiver. Geometry nil means no visibility/occlusion.
286:     /// A point sample nearer than 5 cm, or a receiver within the declared finite-area
287:     /// near-field guard (max of 5 cm and one quarter longest side), is unsupported.
288:     public func evaluate(receiver: Position, normal: UnitDirection, view: UnitDirection,
289:                          material: StudyMaterial, geometry: StudyGeometry? = nil) throws -> StudySurfaceSignal {
290:         guard receiver.isValid, material.isValid, dot3(vec(normal), vec(view)) > 0 else { throw StudyError.invalidSurface }
291:         return try evaluate(receiver: vec(receiver), normal: vec(normal), view: vec(view), material: material, geometry: geometry)
292:     }

294:     /// Internal direct-light evaluator shared by the sphere and mannequin
295:     /// diagnostics. The additional visibility function is CPU-only; the sphere
296:     /// and its Metal oracle share the clipped-area sampling contract.
297:     func evaluate(receiver: SIMD3<Double>, normal: SIMD3<Double>, view: SIMD3<Double>,
298:                   material: StudyMaterial, geometry: StudyGeometry? = nil,
299:                   occlusionTest: ((SIMD3<Double>, SIMD3<Double>, Double) -> Bool)? = nil,
300:                   geometricNormal: SIMD3<Double>? = nil) throws -> StudySurfaceSignal {
301:         if areaGuards.contains(where: { $0.containsNearReceiver(receiver) }) { throw StudyError.invalidSurface }
302:         var total = 0.0, visible = 0.0, xyz = SIMD3<Double>.zero
303:         let shadowOrigin = receiver + (geometricNormal ?? normal) * 1e-5
304:         func accumulate(position: SIMD3<Double>, candela: Double, sample: Sample) throws {
305:             let delta = position - receiver
306:             let r2 = dot3(delta, delta)
307:             guard r2 >= 0.0025 else { throw StudyError.invalidSurface }
308:             let distance = sqrt(r2), wi = delta / distance
309:             // Mesh shading normals must not admit light from below the actual face.
310:             // Analytic sphere and planar receivers retain their original path.
311:             if let geometricNormal, dot3(geometricNormal, wi) <= 0 { return }
312:             let nl = min(1, max(0, dot3(normal, wi)))
313:             if nl == 0 { return }
314:             let emitted = sample.emittingNormal.map { min(1, max(0, -dot3($0, wi))) } ?? 1
315:             if emitted == 0 { return }
316:             let lux = candela * emitted * nl / r2
317:             total += lux
318:             // Offset origin while keeping the shadow ray aimed at the actual source sample.
319:             if geometry != nil || occlusionTest != nil {
320:                 let segment = position - shadowOrigin
321:                 let length = sqrt(dot3(segment, segment))
322:                 let shadowDirection = segment / length
323:                 if geometry?.occludes(from: shadowOrigin, to: shadowDirection, distance: length) == true ||
324:                     occlusionTest?(shadowOrigin, shadowDirection, length) == true { return }
325:             }
326:             visible += lux
327:             let scale = lux * material.response(normal: normal, light: wi, view: view, nl: nl) / 1000
328:             xyz += sample.xyz * scale
329:         }
330:         var index = 0
331:         while index < samples.count {
332:             let sample = samples[index]
333:             guard let areaIndex = sample.areaIndex else {
334:                 try accumulate(position: sample.position, candela: sample.candela, sample: sample)
335:                 index += 1
336:                 continue
337:             }
338:             let area = areaGuards[areaIndex]
339:             // A rectangle is one-sided, so every sample on its back side has
340:             // zero emission. Keep all area guards above this shortcut. The
341:             // additional 5 cm depth and 1 nm roundoff allowance also keep the
342:             // per-sample distance rejection unchanged at its exact boundary;
343:             // coplanar and nearly coplanar receivers use the original path.
344:             if let axis = sample.emittingNormal,
345:                dot3(receiver - area.center, axis) < -(0.05 + 1e-9) {
346:                 index += grid * grid
347:                 continue
348:             }
349:             func plane(_ n: SIMD3<Double>) -> StudyAreaSamplingPlane {
350:                 .init(center: dot3(n, area.center - receiver),
351:                       x: dot3(n, area.right) * area.halfWidth,
352:                       y: dot3(n, area.up) * area.halfHeight)
353:             }
354:             let shading = plane(normal), geometric = geometricNormal.map(plane)
355:             if shading.isFull && (geometric?.isFull ?? true) {
356:                 // Keep both positions and accumulation order on ordinary full faces.
357:                 for offset in 0..<(grid * grid) {
358:                     let full = samples[index + offset]
359:                     try accumulate(position: full.position, candela: full.candela, sample: full)
360:                 }
361:             } else {
362:                 try StudyAreaSliceSampling.forEach(grid: grid, normal: shading, geometric: geometric) { p, weight in
363:                     let position = area.center + area.right * (p.x * area.halfWidth) + area.up * (p.y * area.halfHeight)
364:                     try accumulate(position: position, candela: sample.candela * weight, sample: sample)
365:                 }
366:             }
367:             index += grid * grid
368:         }
369:         return .init(visibleLux: visible, unoccludedLux: total, signalXYZ: .init(x: xyz.x, y: xyz.y, z: xyz.z))
370:     }
371: }

170:     /// lifecycle tests. It can never suppress actual task cancellation.
171:     init(cancellationRequested: @escaping @Sendable () -> Bool = { false }) {
172:         additionalCancellation = cancellationRequested
173:     }
174: 
175:     func render(scene: LightingScene, width: Int = 192, subject: StudySubject = .sphere) throws -> StudyRenderedFrame {
176:         let additionalCancellation = additionalCancellation
177:         let isCancelled = { Task.isCancelled || additionalCancellation() }
178:         guard !isCancelled() else { throw StudyError.cancelled }
179:         let renderWidth = subject == .bust ? min(width, PortraitStudyRenderer.maximumWidth) : width
180:         let key = CacheKey(scene: scene, width: renderWidth, subject: subject)
181:         if let key {
182:             // Check before the cache promotes a hit. A cancelled request must
183:             // neither publish pixels nor change the eviction order.
184:             guard !isCancelled() else { throw StudyError.cancelled }
185:             if let cached = frameCache.value(for: key) { return cached }
186:         }
187:         if subject == .bust {
188:             // The mannequin has additional geometry. Do not send it
189:             // to the sphere-only Metal kernel or claim a validated GPU path.
190:             let image = try PortraitStudyRenderer.render(scene: scene, width: renderWidth,
191:                                                         isCancelled: isCancelled)
192:             guard !isCancelled() else { throw StudyError.cancelled }
193:             return retain(StudyRenderedFrame(image: image, backend: .cpu), for: key)
194:         }
195:         if case .unchecked = state {
196:             do {
197:                 guard let device = MTLCreateSystemDefaultDevice() else { throw MetalStudyError.unavailable }
198:                 let renderer = try MetalStudyRenderer(device: device)
199:                 try renderer.validate(isCancelled: isCancelled)
200:                 guard !isCancelled() else { throw StudyError.cancelled }
201:                 state = .metal(renderer)
202:             } catch let error as StudyError where error == .cancelled {
203:                 // Incomplete validation is not a failed backend. Leave the
204:                 // state unchecked so the next live request can validate it.
205:                 throw error
206:             } catch {
207:                 guard !isCancelled() else { throw StudyError.cancelled }
208:                 selectCPU()
209:             }
210:         }
211:         guard !isCancelled() else { throw StudyError.cancelled }
212:         if case .metal(let renderer) = state, MetalStudyRenderer.supports(scene) {
213:             do {
214:                 let image = try renderer.render(scene: scene, width: width)
215:                 guard !isCancelled() else { throw StudyError.cancelled }
216:                 return retain(StudyRenderedFrame(image: image, backend: .metal), for: key)
217:             } catch let error as StudyError {
218:                 // Validation/cancellation is a scene error, not a backend failure.
219:                 if error != .invalidComputeOutput { throw error }
220:                 guard !isCancelled() else { throw StudyError.cancelled }
221:                 selectCPU()
222:             } catch {
223:                 guard !isCancelled() else { throw StudyError.cancelled }
224:                 selectCPU()
225:             }
226:         }
227:         let image = try ReferenceStudyRenderer.render(scene: scene, width: width, isCancelled: isCancelled)
228:         guard !isCancelled() else { throw StudyError.cancelled }
229:         return retain(StudyRenderedFrame(image: image, backend: .cpu), for: key)
230:     }
231: 
232:     private func selectCPU() {
233:         state = .cpu
234:         // A runtime backend failure invalidates its old frames as well. A later
235:         // hit must not resurrect Metal after the engine has failed closed.
236:         frameCache.removeAll { $0.backend == .metal }
237:     }
238: 
239:     /// Only called after the final cooperative cancellation check. There is no
240:     /// suspension between that check, cache admission and returning the frame.
241:     private func retain(_ frame: StudyRenderedFrame, for key: CacheKey?) -> StudyRenderedFrame {
242:         guard let key else { return frame }
243:         frameCache.insert(frame, for: key, pixelBytes: frame.image.rgba.count,
244:                           maskBytes: frame.image.unsupportedPixelMask.count)
245:         return frame
246:     }
