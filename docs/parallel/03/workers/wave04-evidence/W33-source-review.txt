45:         func interval(origin: SIMD3<Double>, direction: SIMD3<Double>) -> (Double, Double)? {
46:             let o = (origin - center) / radii, d = direction / radii
47:             let a = portraitDot(d, d), b = portraitDot(o, d)
48:             let c = portraitDot(o, o) - 1
49:             let discriminant = b * b - a * c
50:             guard a.isFinite, a > 0, discriminant.isFinite, discriminant >= 0 else { return nil }
51:             let root = sqrt(discriminant)
52:             return ((-b - root) / a, (-b + root) / a)
53:         }

56:                  maximum: Double) -> Hit? {

60:             // Analytic gradient of the ellipsoid; inverse-square scaling is
61:             // essential for non-uniform radii and accurate incidence.
62:             let gradient = (origin + direction * distance - center) / (radii * radii)
63:             guard let normal = portraitUnit(gradient) else { return nil }
64:             return Hit(distance: distance, normal: normal, geometricNormal: normal, region: region)
