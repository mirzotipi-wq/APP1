4: public struct UnitDirection: Equatable, Sendable {
5:     public let x: Double
6:     public let y: Double
7:     public let z: Double
8: 
9:     public init?(x: Double, y: Double, z: Double) {
10:         guard x.isFinite, y.isFinite, z.isFinite else { return nil }
11:         let scale = max(abs(x), max(abs(y), abs(z)))
12:         guard scale > 0 else { return nil }
13:         let a = x / scale, b = y / scale, c = z / scale
14:         let length = (a * a + b * b + c * c).squareRoot()
15:         self.x = a / length; self.y = b / length; self.z = c / length
16:     }

100:     public static func receiverNormal(in scene: LightingScene) -> UnitDirection? {
101:         guard scene.subject.isValid, scene.camera.isValid else { return nil }
102:         return UnitDirection(x: scene.camera.x - scene.subject.x,
103:                              y: scene.camera.y - scene.subject.y,
104:                              z: scene.camera.z - scene.subject.z)

107:     public static func frame(for light: StudioLight, subject: Position) -> EmitterFrame? {
108:         guard light.isValid, light.effectiveOptics.kind.isAimed else { return nil }
109:         return EmitterFrame(source: light.position, target: light.target(subject: subject),
110:                             rollDegrees: light.effectiveOptics.effectiveRollDegrees)
111:     }
112: 
113:     public static func illuminance(for light: StudioLight, in scene: LightingScene) -> Double? {
114:         guard light.isValid, let normal = receiverNormal(in: scene) else { return nil }
115:         let optics = light.effectiveOptics
116:         switch optics.kind {
117:         case .point:
118:             return Photometry.pointIlluminance(candela: light.candela, source: light.position,
119:                                               receiver: scene.subject, receiverNormal: normal)
120:         case .directional:
121:             // Ideal one-sided cosine emitter. Not a hard-edged spotlight or a measured beam.
122:             guard let frame = frame(for: light, subject: scene.subject),
123:                   let ray = UnitDirection(x: scene.subject.x - light.position.x,
124:                                           y: scene.subject.y - light.position.y,
125:                                           z: scene.subject.z - light.position.z),
126:                   let lux = Photometry.pointIlluminance(candela: light.candela, source: light.position,
127:                                                         receiver: scene.subject, receiverNormal: normal) else { return nil }
128:             let n = frame.normal
129:             let emission = min(1, max(0, n.x * ray.x + n.y * ray.y + n.z * ray.z))
130:             return lux * emission
131:         case .rectangle:
132:             guard let frame = frame(for: light, subject: scene.subject),
133:                   let emitter = LambertianRectangle(center: light.position, right: frame.right, up: frame.up,
134:                                                       widthMeters: optics.widthMeters, heightMeters: optics.heightMeters,
135:                                                       onAxisCandela: light.candela) else { return nil }
136:             return Photometry.rectangularIlluminance(emitter: emitter, receiver: scene.subject, receiverNormal: normal)
137:         }

19:             .precision(.fractionLength(fractionDigits))

23:     public static func formattedValue(_ value: Double, unit: String = "",
24:                                       locale: Locale = .current) -> String {
25:         guard value.isFinite else { return unit.isEmpty ? "Unavailable" : "Unavailable " + unit }
26:         let source = decimalText(value, omitLeadingZeroWhenNeeded: false) ?? String(value)
27:         let integerStyle = IntegerFormatStyle<Int>.number.grouping(.never).locale(locale)
28:         let digits = (0...9).map { $0.formatted(integerStyle) }
29:         let minus = (-1).formatted(integerStyle).replacingOccurrences(of: digits[1], with: "")
30:         let number = source.map { character -> String in
31:             if let digit = character.wholeNumberValue { return digits[digit] }
32:             if character == "." { return locale.decimalSeparator ?? "." }
33:             if character == "-" { return minus }
34:             return String(character)
35:         }.joined()
36:         return unit.isEmpty ? number : number + " " + unit
37:     }
