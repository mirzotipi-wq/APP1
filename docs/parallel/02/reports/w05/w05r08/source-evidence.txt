23: /// Adjusts one ideal source's contribution at the camera-facing subject plane.
24: /// This is a model calculation, not a measured lamp setting or image exposure.
25: /// Other lights, occlusion, reflections and real modifier losses are excluded.
26: public enum LightOutputAdjustment {
27:     /// Returns a valid light with only its candela changed. The source and scene
28:     /// remain unchanged, and the light need not already belong to the scene.
29:     /// Positive targets require a supported, nonzero contribution at one candela.
30:     /// Zero explicitly turns the source off without requiring a photometric sample.
31:     /// Invalid inputs and unrepresentable outputs throw; values are never clamped.
32:     public static func light(_ source: StudioLight, targetingLux targetLux: Double,
33:                              in scene: LightingScene) throws -> StudioLight {
34:         guard targetLux.isFinite, targetLux >= 0 else {
35:             throw LightOutputAdjustmentError.invalidTargetLux
36:         }
37:         guard source.isValid else { throw LightOutputAdjustmentError.invalidLight }
38:         guard let scene = try? scene.validated() else {
39:             throw LightOutputAdjustmentError.unsupportedGeometry
40:         }
41: 
42:         var adjusted = source
43:         if targetLux == 0 {
44:             adjusted.candela = 0
45:             return adjusted
46:         }
47: 
48:         // Using a unit source also supports increasing a currently off light;
49:         // scaling its existing zero contribution would be undefined.
50:         adjusted.candela = 1
51:         guard let unitLux = StudioPhotometry.illuminance(for: adjusted, in: scene),
52:               unitLux.isFinite, unitLux > 0 else {
53:             throw LightOutputAdjustmentError.unsupportedGeometry
54:         }
55: 
56:         // Bound the request by the forward model, not the independently
57:         // rounded inverse quotient. Even the next representable lux above
58:         // this output is outside the supported physical range.
59:         var maximum = source
60:         maximum.candela = 100_000
61:         guard maximum.isValid,
62:               let maximumLux = StudioPhotometry.illuminance(for: maximum, in: scene),
63:               maximumLux.isFinite, maximumLux > 0, targetLux <= maximumLux else {
64:             throw LightOutputAdjustmentError.outputOutOfRange
65:         }
66: 
67:         // Preserve an already exact setting before inverting the model.
68:         // Independently rounded division and integration can miss an exactly
69:         // achieved target, including subnormal values and the upper boundary.
70:         if source.candela > 0,
71:            StudioPhotometry.illuminance(for: source, in: scene) == targetLux {
72:             return source
73:         }
74: 
75:         adjusted.candela = targetLux / unitLux
76:         if adjusted.candela <= 0 || !adjusted.isValid {
77:             guard adjusted.candela > maximum.candela else {
78:                 throw LightOutputAdjustmentError.outputOutOfRange
79:             }
80:             // The validated maximum is a separate forward-model candidate.
81:             // It must still achieve the request within the accuracy gate below.
82:             adjusted = maximum
83:         }
84: 
85:         // Extremely small positive outputs can underflow or lose substantial
86:         // precision in finite-emitter integration even when their candela is
87:         // representable. A nonzero result alone does not establish that the
88:         // requested contribution was achieved. Comparing a ratio also avoids
89:         // underflowing an absolute tolerance for subnormal target values.
90:         guard let achievedLux = StudioPhotometry.illuminance(for: adjusted, in: scene),
91:               achievedLux.isFinite, achievedLux > 0,
92:               abs(achievedLux / targetLux - 1) <= 1e-9 else {
93:             throw LightOutputAdjustmentError.outputOutOfRange
94:         }
95:         return adjusted
96:     }
97: }

114:         guard light.isValid, let normal = receiverNormal(in: scene) else { return nil }
115:         let optics = light.effectiveOptics

122:             guard let frame = frame(for: light, subject: scene.subject),

124:                                           y: scene.subject.y - light.position.y,

127:                                                         receiver: scene.subject, receiverNormal: normal) else { return nil }

129:             let emission = min(1, max(0, n.x * ray.x + n.y * ray.y + n.z * ray.z))
