 101     /// Integrates E = L cos(thetaEmitter) cos(thetaReceiver) dA / r^2,
 102     /// with L = on-axis cd / emitter area. Fully front-facing emitters use
 103     /// the same 32x32 midpoint grid as the unoccluded reference renderer.
 104     /// Partially front-facing emitters are clipped to the receiver hemisphere,
 105     /// preventing a narrow contributing strip from disappearing between samples.
 106     /// A clipped rectangle has at most five vertices; 12x12 Gaussian quadrature
 107     /// per triangle bounds this partial-face calculation to 432 evaluations.
 108     /// The sampled renderer remains approximate at grazing incidence.
 109     /// The receiver must still be separated from the emitter plane by at least
 110     /// 5 cm and one quarter of the longest side; nearer configurations abstain.
 111     /// No occlusion, reflection, spectral effects or modifier transmission.
 112     public static func rectangularIlluminance(emitter: LambertianRectangle, receiver: Position,
 113                                               receiverNormal: UnitDirection) -> Double? {
 114         guard receiver.isValid else { return nil }
 115         let toReceiver = vector(receiver) - vector(emitter.center)
 116         let depth = dot(toReceiver, emitter.normal)
 117         let minimumDepth = max(0.05, max(emitter.widthMeters, emitter.heightMeters) / 4)
 118         guard abs(depth) >= minimumDepth else { return nil }
 119         if depth < 0 || emitter.onAxisCandela == 0 { return 0 }
 120 
 121         // Keep the geometric integration at unit intensity when dividing an
 122         // extremely small output among cells would lose its significant bits.
 123         // Apply intensity once, after summation. The ordinary-output path keeps
 124         // its established sample ordering and rounding for renderer parity.
 125         if emitter.onAxisCandela < Double.leastNormalMagnitude * 1024 {
 126             guard let unit = LambertianRectangle(
 127                 center: emitter.center, right: emitter.right, up: emitter.up,
 128                 widthMeters: emitter.widthMeters, heightMeters: emitter.heightMeters,
 129                 onAxisCandela: 1),

 131                                                      receiverNormal: receiverNormal) else { return nil }
 132             return unitLux * emitter.onAxisCandela

 136         let normal = receiverNormal.vector
 137         let nx = dot(right, normal), ny = dot(up, normal)
 138         let offset = dot(toReceiver, normal)
 139         let halfWidth = emitter.widthMeters / 2
 140         let halfHeight = emitter.heightMeters / 2
 141         let corners: [SIMD2<Double>] = [
 142             .init(-halfWidth, -halfHeight), .init(halfWidth, -halfHeight),
 143             .init(halfWidth, halfHeight), .init(-halfWidth, halfHeight)
 144         ]
 145         let vertices = corners.map { p in
 146             IncidenceVertex(point: p, incidence: nx * p.x + ny * p.y - offset)
 147         }
 148         if vertices.allSatisfy({ $0.incidence <= 0 }) { return 0 }
 149         if vertices.allSatisfy({ $0.incidence >= 0 }) {
 150             // Preserve the existing 32x32 sample convention, ordering, and
 151             // normalization for ordinary fully illuminated surfaces.
 152             return fullRectangleMidpoint(emitter: emitter, toReceiver: toReceiver,
 153                                          receiverNormal: receiverNormal)
 154         }
 155         var polygon: [IncidenceVertex] = []
 156         polygon.reserveCapacity(5)
 157         // Sutherland-Hodgman clipping against one affine half-plane. A
 158         // boundary vertex has zero contribution; avoid duplicating it.
 159         for i in vertices.indices {
 160             let a = vertices[(i + vertices.count - 1) % vertices.count]
 161             let b = vertices[i]
 162             if (a.incidence < 0 && b.incidence > 0) || (a.incidence > 0 && b.incidence < 0) {
 163                 let t = a.incidence / (a.incidence - b.incidence)
 164                 polygon.append(.init(point: a.point + (b.point - a.point) * t, incidence: 0))
 165             }
 166             if b.incidence >= 0 { polygon.append(b) }
 167         }
 168         guard polygon.count >= 3 else { return 0 }
 169 
 170         var integral = 0.0
 171         let a = polygon[0]
 172         for i in 1..<(polygon.count - 1) {
 173             let b = polygon[i], c = polygon[i + 1]
 174             let ab = b.point - a.point, ac = c.point - a.point
 175             let determinant = abs(ab.x * ac.y - ab.y * ac.x)
 176             if determinant == 0 { continue }
 177             // Duffy map: p = a + u(b-a) + (1-u)v(c-a).
 178             // The Jacobian is determinant * (1-u).
 179             for (u, wu) in gauss12 {
 180                 let remaining = 1 - u
 181                 for (v, wv) in gauss12 {
 182                     let cWeight = remaining * v
 183                     let aWeight = remaining * (1 - v)
 184                     let p = a.point + ab * u + ac * cWeight
 185                     // The signed projection is affine across each triangle.
 186                     // Its three nonnegative vertex values prevent cancellation
 187                     // from erasing a sliver of grazing-angle illumination.
 188                     let incidence = aWeight * a.incidence + u * b.incidence + cWeight * c.incidence
 189                     let delta = toReceiver - right * p.x - up * p.y
 190                     let r2 = dot(delta, delta)
 191                     integral += determinant * wu * wv * remaining * depth * incidence / (r2 * r2)
 192                 }
 193             }
 194         }
 195         let radiance = emitter.onAxisCandela / (emitter.widthMeters * emitter.heightMeters)
 196         return radiance * integral
 197     }

 199     /// Existing reference-grid integration for fully front-facing receivers.
 200     private static func fullRectangleMidpoint(emitter: LambertianRectangle,
 201                                               toReceiver: SIMD3<Double>,
 202                                               receiverNormal: UnitDirection) -> Double {
 203         let samples = 32
 204         let candelaPerCell = emitter.onAxisCandela / Double(samples * samples)
 205         var lux = 0.0
 206         for j in 0..<samples {
 207             let y = ((Double(j) + 0.5) / Double(samples) - 0.5) * emitter.heightMeters
 208             for i in 0..<samples {
 209                 let x = ((Double(i) + 0.5) / Double(samples) - 0.5) * emitter.widthMeters

 211                 let r2 = dot(delta, delta)
 212                 let direction = delta / r2.squareRoot()
 213                 let emission = min(1, max(0, dot(direction, emitter.normal)))
 214                 let incidence = min(1, max(0, -dot(direction, receiverNormal.vector)))
 215                 lux += candelaPerCell * emission * incidence / r2
 216             }
 217         }

  47     public var subject: Position
  48     public var camera: Position

  57     public init(subject: Position, camera: Position, lights: [StudioLight], cameraLens: CameraLens? = nil, previewSettings: PreviewSettings? = nil, placementRoom: ManualRoom? = nil) {

  59         self.placementRoom = placementRoom

  62     public func validated() throws -> Self {
  63         guard (1...Self.currentSchemaVersion).contains(schemaVersion) else { throw SceneError.unsupportedVersion }
  64         guard lights.count <= 12 else { throw SceneError.invalidScene }
  65         // Version 1 never contained optics; reject ambiguous mislabeled files.
  66         guard schemaVersion != 1 || lights.allSatisfy({ $0.optics == nil }) else { throw SceneError.invalidScene }
  67         // Version 2 adds light optics, version 3 adds a camera lens.
  68         guard schemaVersion >= 3 || cameraLens == nil else { throw SceneError.invalidScene }
  69         guard schemaVersion >= 4 || previewSettings == nil else { throw SceneError.invalidScene }
  70         // Version 5 adds measured rooms and rectangular-emitter roll.
  71         guard schemaVersion >= 5 || (placementRoom == nil && lights.allSatisfy({ $0.optics?.rollDegrees == nil })) else {
  72             throw SceneError.invalidScene
  73         }
  74         guard subject.isValid, camera.isValid, (cameraLens?.isValid ?? true), (previewSettings?.isValid ?? true),
  75               (placementRoom?.isValid ?? true) else { throw SceneError.invalidScene }
  76         guard lights.allSatisfy({ $0.isValid }) else { throw SceneError.invalidLight }
  77         guard Set(lights.map(\.id)).count == lights.count else { throw SceneError.duplicateLight }
  78         var migrated = self
  79         migrated.schemaVersion = Self.currentSchemaVersion
  80         return migrated
  81     }

   9 /// Positive yaw rotates scene +Z toward room +X; scale is always 1:1.

  11     public static let dimensionRange: ClosedRange<Double> = 1...30
  12     public var widthMeters: Double
  13     public var depthMeters: Double
  14     /// Nil means the ceiling height has not been measured.
  15     public var ceilingHeightMeters: Double?

  17     public var subjectZ: Double

  20     public init(widthMeters: Double = 6, depthMeters: Double = 7,

  24         self.ceilingHeightMeters = ceilingHeightMeters

  28     /// Resizing is independent of the subject's location. The editor must
  29     /// require an explicit subject move before accepting a smaller room.

  33     private func dimensionRangePreserving(_ offset: Double) -> ClosedRange<Double> {
  34         guard offset.isFinite, offset >= 0, offset <= Self.dimensionRange.upperBound else {
  35             return Self.dimensionRange
  36         }
  37         return max(Self.dimensionRange.lowerBound, offset)...Self.dimensionRange.upperBound
  38     }
  39 
  40     /// Rotate relative to the current placement, preserving dimensions and
  41     /// subject offsets. Canonical output is in (-180, 180], matching the UI's
  42     /// slider range. Invalid input leaves the entire room unchanged.
  43     @discardableResult
  44     public mutating func rotate(byDegrees degrees: Double) -> Bool {
  45         guard isValid, degrees.isFinite else { return false }
  46         // Reduce the delta before addition so even a very large finite input
  47         // cannot overflow or lose the bounded starting orientation.
  48         var angle = (yawDegrees + degrees.truncatingRemainder(dividingBy: 360))
  49             .truncatingRemainder(dividingBy: 360)
  50         if angle > 180 { angle -= 360 }
  51         if angle <= -180 { angle += 360 }
  52         yawDegrees = angle == 0 ? 0 : angle
  53         return true
  54     }

  60         subjectX.isFinite && subjectZ.isFinite && subjectX >= 0 && subjectX <= widthMeters &&
  61         subjectZ >= 0 && subjectZ <= depthMeters &&
  62         yawDegrees.isFinite && (-360...360).contains(yawDegrees)
